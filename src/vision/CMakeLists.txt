if(NOT DEFINED CMAKE_INSTALL_INCLUDEDIR)
    set(CMAKE_INSTALL_INCLUDEDIR include)
endif()
set(rtctrl_v4l2_default OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(rtctrl_v4l2_default ON)
endif()
option(RTCTRL_ENABLE_V4L2 "Build the Linux V4L2 camera adapter" ${rtctrl_v4l2_default})
if(RTCTRL_ENABLE_V4L2 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "V4L2 requires Linux; disable RTCTRL_ENABLE_V4L2")
endif()
# C-only contracts/options; no dependency on control joint-profile definitions.
add_library(rtctrl_capture_contracts INTERFACE)
target_include_directories(rtctrl_capture_contracts INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/../../include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_features(rtctrl_capture_contracts INTERFACE c_std_11)
if(RTCTRL_ENABLE_SANITIZERS)
    target_compile_options(rtctrl_capture_contracts INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(rtctrl_capture_contracts INTERFACE -fsanitize=address,undefined)
elseif(RTCTRL_ENABLE_TSAN)
    target_compile_options(rtctrl_capture_contracts INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(rtctrl_capture_contracts INTERFACE -fsanitize=thread)
endif()
add_library(rtctrl_capture capture.c)
target_link_libraries(rtctrl_capture PUBLIC rtctrl_capture_contracts)
add_library(rtctrl_capture_synthetic synthetic_capture.c)
target_link_libraries(rtctrl_capture_synthetic PUBLIC rtctrl_capture)
set(capture_targets rtctrl_capture rtctrl_capture_synthetic)
if(RTCTRL_ENABLE_V4L2)
    add_library(rtctrl_vision_v4l2 v4l2_capture.c v4l2_format.c)
    target_link_libraries(rtctrl_vision_v4l2 PUBLIC rtctrl_capture)
    list(APPEND capture_targets rtctrl_vision_v4l2)
endif()
foreach(target IN LISTS capture_targets)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endforeach()
