cmake_minimum_required(VERSION 3.16)
project(rtctrl_audio_aec LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(RKAUDIO_ROOT "" CACHE PATH "BSP external/common_algorithm/audio/rkaudio_algorithms")
if(NOT EXISTS "${RKAUDIO_ROOT}/include/rkaudio_preprocess.h")
    message(FATAL_ERROR "Set RKAUDIO_ROOT to BSP rkaudio_algorithms")
endif()
add_library(rtctrl_aec SHARED aec_bridge.cc)
target_include_directories(rtctrl_aec PRIVATE "${RKAUDIO_ROOT}/include")
target_link_libraries(rtctrl_aec PRIVATE "${RKAUDIO_ROOT}/lib64/librkaudio_vqe.so" "${RKAUDIO_ROOT}/lib64/librkaudio_common.so" m dl)
set_target_properties(rtctrl_aec PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
add_executable(aec_probe aec_probe.cc)
target_link_libraries(aec_probe PRIVATE rtctrl_aec m dl)
target_link_options(aec_probe PRIVATE "-Wl,-rpath-link,${RKAUDIO_ROOT}/lib64")
set_target_properties(aec_probe PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN")
