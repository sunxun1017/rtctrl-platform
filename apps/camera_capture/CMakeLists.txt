cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(camera_capture LANGUAGES C)
    add_subdirectory(../../src/vision ${CMAKE_CURRENT_BINARY_DIR}/vision)
endif()
# The shared consumer has no adapter includes or selected-backend macros.
add_library(rtctrl_capture_cli STATIC capture_cli.c)
target_link_libraries(rtctrl_capture_cli PUBLIC rtctrl_capture)
target_compile_options(rtctrl_capture_cli PRIVATE -Wall -Wextra -Wpedantic)
add_executable(rtctrl_camera_synthetic synthetic_main.c)
target_link_libraries(rtctrl_camera_synthetic PRIVATE rtctrl_capture_cli rtctrl_capture_synthetic)
if(TARGET rtctrl_vision_v4l2)
    add_executable(rtctrl_camera_capture main.c)
    target_link_libraries(rtctrl_camera_capture PRIVATE rtctrl_capture_cli rtctrl_vision_v4l2)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    install(TARGETS rtctrl_camera_synthetic RUNTIME DESTINATION bin)
    if(TARGET rtctrl_camera_capture)
        install(TARGETS rtctrl_camera_capture RUNTIME DESTINATION bin)
    endif()
endif()
