rtctrl_module_api(inference)
target_compile_features(rtctrl_inference_api INTERFACE cxx_std_17)
if(RTCTRL_ENABLE_SANITIZERS)
    target_compile_options(rtctrl_inference_api INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(rtctrl_inference_api INTERFACE -fsanitize=address,undefined)
elseif(RTCTRL_ENABLE_TSAN)
    target_compile_options(rtctrl_inference_api INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(rtctrl_inference_api INTERFACE -fsanitize=thread)
endif()

target_link_libraries(rtctrl_inference PUBLIC rtctrl_inference_api)
target_compile_options(rtctrl_inference PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
