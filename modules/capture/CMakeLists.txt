rtctrl_module_api(capture)
target_compile_features(rtctrl_capture_api INTERFACE c_std_11)
if(RTCTRL_ENABLE_SANITIZERS)
    target_compile_options(rtctrl_capture_api INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(rtctrl_capture_api INTERFACE -fsanitize=address,undefined)
elseif(RTCTRL_ENABLE_TSAN)
    target_compile_options(rtctrl_capture_api INTERFACE -fsanitize=thread -fno-omit-frame-pointer)
    target_link_options(rtctrl_capture_api INTERFACE -fsanitize=thread)
endif()
rtctrl_module_library(rtctrl_capture src/capture.c)
target_link_libraries(rtctrl_capture PUBLIC rtctrl_capture_api)
target_compile_options(rtctrl_capture PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
